val txtShake = sc.textFile("shake.txt")   // RDD con una linea del archivo por elemento (el original usa repos.txt, que no existe)
txtShake.collect                          // accion: trae todas las lineas al driver y las imprime

val query = txtShake.flatMap(line => line.split(",")).map(word => (word,1)).reduceByKey(_ + _)   // parte por comas, arma (trozo,1) y suma por trozo
query.toDebugString                       // muestra el linaje: la cadena de transformaciones desde shake.txt
query.cache()                             // marca el RDD para guardarlo en memoria cuando se calcule

query.saveAsTextFile("output_shake_")     // accion: ejecuta todo y guarda el resultado en la carpeta output_shake_
query.collect                             // accion: trae los pares (trozo, cantidad) al driver (sale del cache)

// Declarar Variable
val x = 6 + 4                 // variable inmutable (val): x = 10, tipo Int deducido

	val x = 3 + 2             // la ultima linea es una declaracion -> el bloque vale () (Unit, "nada")
})

print({
	val x = 3 + 2             // x = 5 solo dentro del bloque
        x + 7                 // ultima linea -> el bloque vale 12 y eso se imprime
})

(x :Int) => x + 2             // funcion anonima (lambda): recibe un Int y le suma 2; sola no hace nada

val masDos = (x: Int) => x + 2   // guarda la lambda en una variable para poder llamarla
println(masDos(3))               // la llama con 3 -> imprime 5

println(mulCinco(7))             // imprime 35
mulCinco(5)                      // la consola muestra res: Int = 25

def mult2(x: Int, y: Int)(aux: Int):    // def = funcion con nombre; dos listas de parametros (currying)
Int = (x * y)*aux                       // devuelve Int: (x*y)*aux
println(mult2(3,5)(2))                  // se llama con dos pares de parentesis -> (3*5)*2 = 30

def mult3(x: Int, y: Int):              // funcion con nombre de dos parametros
Int = (x * y)                           // devuelve x*y

object OpeRand{
	private var ale = 1                     // contador que empieza en 1
	def porDos():Int = {ale *= 2; ale}      // multiplica por 2 y devuelve el valor (; separa instrucciones en una linea)
	def masDos():Int = {ale += 2; ale}      // suma 2 y devuelve el valor
}

val ab1: Int = OpeRand.porDos()   // 1*2 = 2
val ab2: Int = OpeRand.masDos()   // 2+2 = 4
val ab1: Int = OpeRand.porDos()   // 4*2 = 8 (sigue desde el valor anterior: hay un solo OpeRand)

// Tarea: pasar el object OpeRand a class (cada instancia con new tiene su propio ale)
class OpeRandC{
	private var ale = 1                     // cada objeto creado con new tiene su propio ale
	def porDos():Int = {ale *= 2; ale}
	def masDos():Int = {ale += 2; ale}
}
val op1 = new OpeRandC    // primera instancia (ale = 1)
val op2 = new OpeRandC    // segunda instancia, independiente (ale = 1)
op1.porDos()              // 2
op1.masDos()              // 4
